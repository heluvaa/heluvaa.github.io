    <script>
        function extractCookie() {
            const curlText = document.getElementById('curlInput').value;
            const outputBox = document.getElementById('cookieOutput');
            
            // Regex untuk mendeteksi cookie dari string cURL (mendukung cn_session atau format umum)
            const regex = /cookie:\s*['"]?(.*?)['"]?\b/i;
            const match = curlText.match(regex);
            
            if (match && match[1]) {
                let cleanCookie = match[1].trim().replace(/^['"]|['"]$/g, '');
                outputBox.value = cleanCookie;
            } else {
                // Alternatif pencarian jika format header menggunakan argumen terpisah
                const altRegex = /-H\s+['"].*?cookie:\s*(.*?)['"]/i;
                const altMatch = curlText.match(altRegex);
                if (altMatch && altMatch[1]) {
                    outputBox.value = altMatch[1].trim().replace(/^['"]|['"]$/g, '');
                } else {
                    alert('Cookie tidak ditemukan dalam teks cURL tersebut!');
                    outputBox.value = '';
                }
            }
        }

        function clearFields() {
            document.getElementById('curlInput').value = '';
            document.getElementById('cookieOutput').value = '';
        }

        function copyToClipboard() {
            const outputBox = document.getElementById('cookieOutput');
            if (!outputBox.value) {
                alert('Tidak ada cookie yang bisa disalin!');
                return;
            }
            
            outputBox.select();
            outputBox.setSelectionRange(0, 99999); // Untuk perangkat seluler
            
            navigator.clipboard.writeText(outputBox.value).then(() => {
                alert('Cookie berhasil disalin ke clipboard!');
            }).catch(err => {
                alert('Gagal menyalin cookie.');
            });
        }
    </script>
